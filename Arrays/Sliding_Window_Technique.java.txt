import java.util.*;
class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] arr = new int[n];
        for(int i = 0; i < n; i++){
            arr[i] = sc.nextInt();
        }
        int k = 3;
        int WindowSum = 0;
        for(int i = 0; i < k; i++){
            WindowSum += arr[i];
        }
        int MaxSum = WindowSum;
        for(int i = k; i < n; i++){
            WindowSum = WindowSum - arr[i-k] + arr[i];
            MaxSum = Math.max(WindowSum, MaxSum);
        }
        System.out.print(MaxSum);
    }
}
